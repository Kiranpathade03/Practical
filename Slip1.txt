✅ 4️⃣ Queries
a️⃣ Display area wise property details
db.properties.find({}, { area: 1, rate: 1, _id: 0 });


 b Display property owned by "Mr. Patil" having minimum rate
db.properties.aggregate([
  { $match: { owner_id: 1 }},
  { $sort: { rate: 1 }},
  { $limit: 1 }
]);

c Give details of owner whose property is at Nashik
db.properties.aggregate([
  { $match: { area: "Nashik" }},
  {
    $lookup: {
      from: "owners",
      localField: "owner_id",
      foreignField: "_id",
      as: "OwnerDetails"
    }
  }
]);

d Display area of property whose rate is less than 100000
db.properties.find(
  { rate: { $lt: 100000 }},
  { area: 1, rate: 1, _id: 0 }
);
