Create a Node.js file that will convert the output "Full Stack!" into reverse string.

ANS 
const str = "Full Stack!";

const reversedString = str.split("").reverse().join("");

console.log("Original String:", str);
console.log("Reversed String:", reversedString);


Q.2) Using node js create a web page to read two file names from user and append contents of first 
file into second file.


const fs = require('fs');
const readline = require('readline');

const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout
});

rl.question("Enter first file name: ", function(file1) {

    rl.question("Enter second file name: ", function(file2) {

        // Read content of first file
        fs.readFile(file1, 'utf8', (err, data) => {

            if (err) {
                console.log("Error reading first file");
                rl.close();
                return;
            }

            // Append content into second file
            fs.appendFile(file2, data, (err) => {

                if (err) {
                    console.log("Error appending file");
                } else {
                    console.log("Content appended successfully.");
                }

                rl.close();
            });
        });
    });
});
