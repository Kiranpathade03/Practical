** Write a python program using mean end analysis algorithmproblem of transforming a string of lowercase letters into another string **

 

def means_end_analysis(start, goal):
    current = start
    step = 0

    print("Start String :", start)
    print("Goal  String :", goal)
    print("----------------------------------")

    while current != goal:
        step += 1
        print(f"Step {step}:")
        print(" Current string:", current)

         
        if len(current) < len(goal):
           
            op = "Insert"
            pos = len(current)
            char = goal[pos]
            current = current + char
            print(f" Operation    : {op} '{char}' at position {pos}")

        elif len(current) > len(goal):
            
            op = "Delete"
            char = current[-1]
            current = current[:-1]
            print(f" Operation    : {op} '{char}' from last position")

        else:
            
            i = 0
            while i < len(current) and current[i] == goal[i]:
                i += 1

            if i < len(current):
                op = "Substitute"
                old_char = current[i]
                new_char = goal[i]
                current = current[:i] + new_char + current[i+1:]
                print(f" Operation    : {op} '{old_char}' -> '{new_char}' at position {i}")
            else:
              
                break

        print(" New string   :", current)
        print("----------------------------------")

    print("Final String  :", current)
    if current == goal:
        print("Result        : Goal reached!")
    else:
        print("Result        : Could not reach goal.")


 
start_str = input("Enter start string (lowercase): ")
goal_str = input("Enter goal string  (lowercase): ")

means_end_analysis(start_str, goal_str)


Q.2)Write a Python program to solve water jug problem. Two jugs with capacity 4 gallon and 3 gallon are given with unlimited water supply respectively. The target is to achieve 2 gallon of water in second jug. ***


from collections import deque

def water_jug():
   
    start_state = (0, 0)
    goal = 2

    
    queue = deque([start_state])

 
    visited = set()
    visited.add(start_state)

    print("Steps to reach the goal:\n")

    while queue:
        a, b = queue.popleft()
        print(f"Current State: ({a} gallons, {b} gallons)")
 
        if b == goal:
            print("\n🎯 Goal Reached! -> Final State:", (a, b))
            return

        
        next_states = [

            (4, b),   # Fill Jug A
            (a, 3),   # Fill Jug B
            (0, b),   # Empty Jug A
            (a, 0),   # Empty Jug B

             
            (max(0, a - (3 - b)), min(3, b + a)),
           
            (min(4, a + b), max(0, b - (4 - a)))
        ]

        
        for state in next_states:
            if state not in visited:
                visited.add(state)
                queue.append(state)

 
water_jug()
